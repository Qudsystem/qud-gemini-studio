## هدفنا بالظبط
- تطبيقك = واجهة للمستخدم.
- تطبيق AI Studio = عقل / منطق / برومبتات.
- نبعت له صور وتعليمات ونرجّع نتيجة.

AI Studio نفسه مش بيدي API، فهنخترع **طبقة ذكية حوالينه** تخليك تستفيد من نفس المنطق من غير ما تلمس واجهته كل مرة.

## طريقة مبتكرة بس عملية: “تشغيل التطبيق كـ Service داخل متصفح”

الفكرة:  
نحوّل AI Studio app إلى **خدمة محلية** بتشتغل جوه متصفحك/سيرفرك، وتطبيقك يكلمها بشكل غير مباشر.

### شكل السيستم
1. **متصفح أو Chrome Headless (Playwright / Puppeteer)**:
   - يفتح صفحة AI Studio بتاعتك (مرة واحدة).
   - يفضّل يبقى على سيرفر أو جهاز شغال 24/7.

2. **بوت تحكم (Automation Script)**:
   - مكتوب بـ Node.js / Python باستخدام Puppeteer أو Playwright.
   - يستقبل Requests من تطبيقك (REST أو WebSocket).
   - جوه المتصفح:
     - يرفع الصورة.
     - يكتب البرومبت.
     - يضغط على زر التشغيل في AI Studio.
     - ينتظر النتيجة (صورة أو رابط).
   - يرجّع النتيجة كـ URL أو Base64 لتطبيقك.

3. **تطبيقك**:
   - يبعت JSON بسيط لسكربت الأتمتة:
     - `{ imageUrl, prompt, mode: "generate" | "edit" }`
   - يستقبل صورة جاهزة ويعرضها للمستخدم.

### ليه الطريقة دي مبتكرة ومختلفة عن “الحلول الزفت”؟
- أنت فعلياً **بتستخدم نفس التطبيق** اللي عملته على AI Studio، بنفس المنطق، بدون ما تعيد كتابة البرومبتات أو الفلو.
- تطبيقك شايف حاجة واحدة: **API داخلي بتاعك**، بينما تحت الكبوت في متصفح شغال بيشغل AI Studio أوتوماتيك.
- تقدر تزوّد عليها أي منطق (Caching، Queue، Limits…).

### خطوات التنفيذ على مستوى عالٍ

#### 1) سكربت أتمتة (Node + Playwright مثلاً)
- يستخدم Playwright:
  - يفتح `https://aistudio.google.com/...`.
  - يسجّل الدخول بحسابك مرة واحدة (تخزين Cookies / Session).
  - يحافظ على التبويب مفتوح.

#### 2) REST API فوقه
- Express server مثلاً:
  - `POST /process-image`
    - Body: `{ imageUrl, prompt, mode }`
- جوه الـ handler:
  - يستدعي دالة `runAIStep(imageUrl, prompt, mode)` اللي بتستخدم Playwright عشان تتعامل مع صفحة AI Studio:
    - ترفع الصورة (لو mode = edit).
    - تكتب البرومبت.
    - تضغط زر التشغيل.
    - تستخرج الرابط/الصورة الناتجة من DOM.
  - يرجّعها للـ client.

#### 3) تطبيقك الأمامي
- يستدعي `/process-image` بدل ما يتعامل مع Google مباشرة.
- ما يهموش أصلاً إن اللي وراء الـ endpoint ده متصفح + AI Studio.

### ملاحظات مهمة
- دي طريقة “خارج الصندوق” وتعتمد على:
  - استقرار واجهة AI Studio (أي تغيير كبير ممكن يكسّر الأتمتة).
  - تشغيل متصفح أو Headless browser دائماً.
- قانونياً وأخلاقياً:
  - لازم تراجع شروط Google AI Studio وGemini API (بعضهم يمنع الأتمتة غير المصرح بها أو الاستهلاك التجاري بدون API الرسمي).[1][2]

لو موافق نمشي في الاتجاه ده، الخطوة الجاية:
- تقول:
  - تفضّل **Node.js** ولا **Python** للحته دي؟
- بعدها يتم بناء:
  - Skeleton مشروع:
    - سكربت Playwright لفتح وتهيئة الـ App.
    - REST API endpoint.
    - دالة `runAIStep` pseudo-code أو كود فعلي.

// server.js
const express = require('express');
const cors = require('cors');
const { chromium } = require('playwright');
const app = express();

app.use(cors());
app.use(express.json());

// 1. إنشاء جلسة متصفح واحدة
let browser;
let page;

async function initializeBrowser() {
    browser = await chromium.launch({ 
        headless: false, // مهم: false للتغلب على الكابتشا
        args: [
            '--disable-blink-features=AutomationControlled',
            '--disable-dev-shm-usage',
            '--no-sandbox'
        ]
    });
    
    const context = await browser.newContext({
        userAgent: 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
        viewport: { width: 1280, height: 720 }
    });
    
    page = await context.newPage();
    
    // 2. الذهاب إلى AI Studio وتسجيل الدخول
    await page.goto('https://aistudio.google.com/');
    
    // انتظار تسجيل الدخول اليدوي (مرة واحدة)
    console.log('⚠️ قم بتسجيل الدخول يدوياً في المتصفح الذي سيظهر...');
    await page.waitForSelector('textarea', { timeout: 120000 });
    
    console.log('✅ تم تسجيل الدخول بنجاح');
    return page;
}

// 3. API لمعالجة الصور
app.post('/api/process', async (req, res) => {
    try {
        const { imageUrl, prompt, mode } = req.body;
        
        // تأكد من تهيئة الصفحة
        if (!page) {
            await initializeBrowser();
        }
        
        // تنفيذ العملية على AI Studio
        const result = await processOnAIStudio(imageUrl, prompt, mode);
        
        res.json({
            success: true,
            resultUrl: result,
            timestamp: new Date().toISOString()
        });
        
    } catch (error) {
        console.error('Error:', error);
        res.status(500).json({
            success: false,
            error: error.message
        });
    }
});

async function processOnAIStudio(imageUrl, prompt, mode) {
    // التنقل إلى صفحة الصورة
    await page.goto('https://aistudio.google.com/app/prompts/image-editor');
    
    // رفع الصورة
    if (mode === 'edit') {
        const fileInput = await page.$('input[type="file"]');
        await fileInput.setInputFiles(await downloadImage(imageUrl));
        
        await page.waitForTimeout(3000);
    }
    
    // كتابة البرومبت
    await page.fill('textarea', prompt);
    
    // الضغط على زر التشغيل
    await page.click('button:has-text("Generate")');
    
    // انتظار النتيجة
    await page.waitForSelector('img[src*="generated"]', { timeout: 60000 });
    
    // استخراج رابط الصورة
    const resultUrl = await page.$eval('img[src*="generated"]', img => img.src);
    
    return resultUrl;
}

// مساعدة: تحميل الصورة
async function downloadImage(url) {
    const response = await fetch(url);
    const buffer = await response.buffer();
    return buffer;
}

// 4. تشغيل السيرفر
const PORT = process.env.PORT || 3000;
app.listen(PORT, async () => {
    console.log(`🚀 Server running on port ${PORT}`);
    console.log('Initializing browser session...');
    await initializeBrowser();
});

        // طريقة بديلة: سحب وإفلات
        await this.dragAndDropImage(tempPath);
      }
      
      // تنظيف الملف المؤقت
      setTimeout(() => fs.unlink(tempPath).catch(() => {}), 5000);
      
      return tempPath;
    } catch (error) {
      throw new Error(`فشل رفع الصورة: ${error.message}`);
    }
  }

  // ✍️ 8. كتابة البرومبت
  async typePrompt(prompt) {
    await page.waitForSelector('textarea', { timeout: 10000 });
    
    // مسح أي نص موجود
    await page.click('textarea', { clickCount: 3 });
    await page.keyboard.press('Backspace');
    
    // كتابة البرومبت حرفاً حرفاً لمحاكاة الكتابة البشرية
    for (let char of prompt) {
      await page.type('textarea', char, { delay: Math.random() * 30 + 20 });
    }
    
    await page.waitForTimeout(1000);
  }

  // ⚡ 9. النقر على زر التوليد
  async clickGenerate() {
    // البحث عن الزر بمختلف الأسماء المحتملة
    const buttonSelectors = [
      'button:has-text("Generate")',
      'button:has-text("إنشاء")',
      'button[aria-label*="Generate"]',
      'button[data-testid*="generate"]',
      'div[role="button"]:has-text("Generate")'
    ];
    
    for (const selector of buttonSelectors) {
      try {
        const button = await page.$(selector);
        if (button) {
          await button.click({ delay: Math.random() * 100 + 50 });
          logger.info(`✅ تم النقر على الزر: ${selector}`);
          return;
        }
      } catch (e) {
        continue;
      }
    }
    
    throw new Error('لم يتم العثور على زر التوليد');
  }

  // ⏳ 10. انتظار النتيجة
  async waitForResult(timeout = 60000) {
    const startTime = Date.now();
    
    while (Date.now() - startTime < timeout) {
      try {
        // البحث عن الصورة الناتجة
        const resultSelectors = [
          'img[src*="generated"]',
          'img[src*="output"]',
          'div[class*="result"] img',
          'img[alt*="generated"]'
        ];
        
        for (const selector of resultSelectors) {
          const img = await page.$(selector);
          if (img) {
            const src = await img.getAttribute('src');
            if (src && (src.startsWith('http') || src.startsWith('data:'))) {
              logger.info(`✅ تم العثور على النتيجة: ${src.substring(0, 50)}...`);
              return src;
            }
          }
        }
        
        // التحقق من وجود رسالة خطأ
        const errorSelectors = [
          'div[class*="error"]',
          'div:has-text("error")',
          'div:has-text("فشل")',
          'div[role="alert"]'
        ];
        
        for (const selector of errorSelectors) {
          const error = await page.$(selector);
          if (error) {
            const errorText = await error.textContent();
            if (errorText.includes('error') || errorText.includes('فشل')) {
              throw new Error(`AI Studio Error: ${errorText}`);
            }
          }
        }
        
        await page.waitForTimeout(2000);
        
      } catch (error) {
        if (error.message.includes('AI Studio Error')) {
          throw error;
        }
      }
    }
    
    throw new Error('انتهت مهلة انتظار النتيجة');
  }

  // 💾 11. تحميل النتيجة
  async downloadResult(imageUrl) {
    try {
      if (imageUrl.startsWith('data:')) {
        // Base64 image
        const base64Data = imageUrl.split(',')[1];
        return Buffer.from(base64Data, 'base64');
      } else {
        // URL image - تحميل عبر axios
        const response = await axios.get(imageUrl, {
          responseType: 'arraybuffer',
          headers: {
            'Referer': 'https://aistudio.google.com/',
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'
          }
        });
        return Buffer.from(response.data, 'binary');
      }
    } catch (error) {
      logger.error(`❌ فشل تحميل النتيجة: ${error.message}`);
      
      // محاولة التقاط لقطة شاشة كبديل
      const screenshot = await page.screenshot({ type: 'jpeg', quality: 90 });
      return screenshot;
    }
  }

  // 🔄 12. استعادة الجلسة
  async recoverSession() {
    logger.warn('🔄 محاولة استعادة الجلسة...');
    
    try {
      // إغلاق الصفحة الحالية
      if (page) {
        await page.close().catch(() => {});
      }
      
      // إنشاء صفحة جديدة
      page = await context.newPage();
      
      // إعادة تحميل AI Studio
      await page.goto('https://aistudio.google.com/', {
        waitUntil: 'networkidle',
        timeout: 30000
      });
      
      // التحقق من تسجيل الدخول
      const loggedIn = await this.checkLoginStatus();
      
      if (!loggedIn) {
        throw new Error('فشل استعادة الجلسة - يلزم تسجيل دخول جديد');
      }
      
      logger.info('✅ تم استعادة الجلسة بنجاح');
      return true;
      
    } catch (error) {
      logger.error(`❌ فشل استعادة الجلسة: ${error.message}`);
      
      // تنظيف كامل وإعادة تهيئة
      await this.cleanup();
      await this.initialize();
      
      return false;
    }
  }

    }
  }
});

app.post('/api/process', upload.single('image'), async (req, res) => {
  try {
    await rateLimiter.consume(req.ip);
    
    if (!isInitialized) {
      return res.status(400).json({
        success: false,
        error: 'النظام غير مهيأ. يرجى استدعاء /api/initialize أولاً'
      });
    }
    
    const { prompt, mode = 'generate', socketId } = req.body;
    
    if (!prompt) {
      return res.status(400).json({
        success: false,
        error: 'البرومبت مطلوب'
      });
    }
    
    let imageBuffer = null;
    if (req.file) {
      imageBuffer = req.file.buffer;
    }
    
    const result = await aiAutomation.processImage({
      imageBuffer,
      prompt,
      mode,
      socketId
    });
    
    if (result.success && result.imageBuffer) {
      // إرجاع الصورة كـ base64
      const base64Image = result.imageBuffer.toString('base64');
      const dataUrl = `data:image/jpeg;base64,${base64Image}`;
      
      res.json({
        success: true,
        imageUrl: dataUrl,
        timestamp: result.timestamp
      });
    } else {
      res.json(result);
    }
    
  } catch (error) {
    if (error instanceof Error) {
      res.status(500).json({ success: false, error: error.message });
    } else {
      res.status(429).json({ success: false, error: 'تم تجاوز معدل الطلبات المسموح به' });
    }
  }
});

app.post('/api/batch', async (req, res) => {
  // معالجة دفعات من الصور
  const { prompts, mode } = req.body;
  
  if (!Array.isArray(prompts) || prompts.length > 5) {
    return res.status(400).json({
      success: false,
      error: 'يجب أن تكون prompts مصفوفة وبحد أقصى 5 عناصر'
    });
  }
  
  const results = [];
  for (const prompt of prompts) {
    try {
      const result = await aiAutomation.processImage({ prompt, mode });
      results.push(result);
    } catch (error) {
      results.push({ success: false, error: error.message });
    }
  }
  
  res.json({ success: true, results });
});

    });

    if (process.env.AUTO_INITIALIZE === 'true') {
      setTimeout(async () => {

          await aiAutomation.initialize();

      }, 5000);
    }
    
  } catch (error) {
    logger.error(`❌ فشل تشغيل السيرفر: ${error.message}`);
    process.exit(1);
  }
}

        function initWebSocket() {
            const socketUrl = window.location.hostname === 'localhost' 
                ? 'http://localhost:3000' 
                : window.location.origin;
            
            socket = io(socketUrl);
            
            socket.on('connect', () => {
                updateStatus(true, 'متصل');
                showNotification('تم الاتصال بالسيرفر بنجاح', 'success');
            });
            
            socket.on('disconnect', () => {
                updateStatus(false, 'غير متصل');
                showNotification('فقد الاتصال بالسيرفر', 'error');
            });
            
            socket.on('progress', (data) => {
                updateProgress(data.message, data.progress);
            });
            
            socket.on('status', (data) => {
                if (data.initialized) {
                    elements.statusDot.classList.add('active');
                    elements.statusText.textContent = 'جاهز للعمل';

        async function initializeSystem() {
            elements.initializeBtn.disabled = true;

                elements.initializeBtn.disabled = false;

        // معالجة الصورة
        async function processImage(mode) {
            if (!elements.promptInput.value.trim()) {
                showNotification('يرجى كتابة وصف للصورة', 'error');
                return;
            }
            
            if (mode === 'edit' && !selectedFile) {
                showNotification('يرجى اختيار صورة أولاً', 'error');
                return;
            }
            
            // تعطيل الأزرار
            elements.generateBtn.disabled = true;
            elements.editBtn.disabled = true;
            
            // إظهار شريط التقدم
            updateProgress('جاري إرسال الطلب...', 10);
            
            const startTime = Date.now();
            
            try {
                const formData = new FormData();
                formData.append('prompt', elements.promptInput.value);
                formData.append('mode', mode);
                formData.append('socketId', socket.id);
                
                if (selectedFile && mode === 'edit') {
                    formData.append('image', selectedFile);
                }
                
                const response = await fetch('/api/process', {
                    method: 'POST',
                    body: formData
                });
                
                const data = await response.json();
                
                if (data.success) {
                    // حساب الوقت المنقضي
                    const elapsedTime = (Date.now() - startTime) / 1000;
                    
                    // تحديث الصورة الناتجة
                    generatedImageUrl = data.imageUrl;
                    elements.resultImage.src = data.imageUrl;
                    elements.resultSection.style.display = 'block';
                    elements.emptyResult.style.display = 'none';
                    
                    // تحديث المعلومات
                    elements.generationTime.textContent = elapsedTime.toFixed(1);
                    elements.imageSize.textContent = '1024×1024';
                    elements.generationDate.textContent = new Date().toLocaleString('ar-SA');
                    
                    // تحديث الإحصائيات
                    if (mode === 'generate') {
                        stats.generated++;
                        elements.statGenerated.textContent = stats.generated;
                    } else {
                        stats.edited++;
                        elements.statEdited.textContent = stats.edited;
                    }
                    
                    stats.totalTime += elapsedTime;
                    stats.successCount++;
                    stats.totalAttempts++;
                    
                    const avgTime = (stats.totalTime / stats.successCount).toFixed(1);
                    const successRate = ((stats.successCount / stats.totalAttempts) * 100).toFixed(1);
                    
                    elements.statTime.textContent = avgTime;
                    elements.statSuccess.textContent = `${successRate}%`;
                    
                    showNotification(`تم ${mode === 'generate' ? 'توليد' : 'تعديل'} الصورة بنجاح!`, 'success');
                } else {
                    showNotification(`فشل المعالجة: ${data.error}`, 'error');
                }
            } catch (error) {
                showNotification('خطأ في الاتصال بالسيرفر', 'error');
            } finally {
                // إعادة تمكين الأزرار
                elements.generateBtn.disabled = false;
                elements.editBtn.disabled = !selectedFile;
                
                // إخفاء شريط التقدم
                setTimeout(() => {
                    elements.progressContainer.style.display = 'none';
                }, 1000);
            }
        }

        function downloadImage() {

                return;

            const link = document.createElement('a');
            link.href = generatedImageUrl;
            link.download = `ai-generated-${Date.now()}.jpg`;

            link.click();
            document.body.removeChild(link);
            
            showNotification('تم بدء تحميل الصورة', 'success');
        }
        
        // استخدام الصورة كمدخل
        function useAsInput() {
            if (!generatedImageUrl) return;
            
            // تحويل Data URL إلى Blob
            fetch(generatedImageUrl)
                .then(res => res.blob())
                .then(blob => {
                    selectedFile = new File([blob], `input-${Date.now()}.jpg`, { type: 'image/jpeg' });
                    
                    // عرض الصورة
                    elements.imagePreview.src = generatedImageUrl;
                    elements.imagePreview.style.display = 'block';
                    elements.removeImage.style.display = 'flex';
                    elements.editBtn.disabled = false;
                    elements.editBtn.innerHTML = '<i class="fas fa-edit"></i> تعديل هذه الصورة';
                    
                    showNotification('تم استخدام الصورة كمدخل جديد', 'success');
                });
        }
        
        // مشاركة الصورة
        function shareImage() {
            if (!generatedImageUrl) {
                showNotification('لا توجد صورة للمشاركة', 'error');
                return;
            }
            
            if (navigator.share) {
                navigator.share({
                    title: 'صورة مولدة بالذكاء الاصطناعي',
                    text: 'شاهد هذه الصورة المولدة بالذكاء الاصطناعي!',
                    url: generatedImageUrl
                });
            } else {
                // نسخ الرابط
                navigator.clipboard.writeText(generatedImageUrl)
                    .then(() => {
                        showNotification('تم نسخ رابط الصورة', 'success');
                    })
                    .catch(() => {
                        prompt('انسخ الرابط:', generatedImageUrl);
                    });
            }
        }

                    elements.statTime.textContent = avgTime;
                    elements.statSuccess.textContent = `${successRate}%`;
                }
            }
        }

            localStorage.setItem('aiStudioStats', JSON.stringify(stats));

    const backupPath = `sessions/backup/cookies-${Date.now()}.json`;
    await fs.writeFile(backupPath, JSON.stringify(cookies, null, 2));
  } catch (error) {
    logger.error('فشل النسخ الاحتياطي للجلسة');
  }
}, 3600000); // كل ساعة

class MultiUserManager {
  constructor() {
    this.sessions = new Map();
  }
  
  async getUserSession(userId) {
    if (!this.sessions.has(userId)) {
      const session = await this.createNewSession(userId);
      this.sessions.set(userId, session);
    }
    return this.sessions.get(userId);
  }
}
